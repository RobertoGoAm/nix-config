* modules home-manager features development work-vdi

Keep the standalone work editor installer current on every activation.

#+begin_src nix
{
  config,
  lib,
  pkgs,
  ...
}:
let
  export = pkgs.callPackage ../../../../pkgs/work-vdi.nix { };
  checkout = "${config.home.homeDirectory}/nix-config";
  copy = pkgs.writeShellApplication {
    name = "work-vdi-export";
    runtimeInputs = [ pkgs.coreutils ];
    text = ''
      destination=${lib.escapeShellArg "${checkout}/portable/work-vdi"}
      case "''${1:-}" in
        --output)
          if [ "$#" -ne 2 ]; then
            echo "usage: work-vdi-export [--output DIR]" >&2
            exit 2
          fi
          destination="$2"
          ;;
        --help|-h)
          echo "usage: work-vdi-export [--output DIR]"
          echo "Copy the work editor files generated by the current Nix generation."
          exit 0
          ;;
        "") ;;
        *) echo "usage: work-vdi-export [--output DIR]" >&2; exit 2 ;;
      esac
      mkdir -p "$destination/helpers"
      for source in ${export}/* ${export}/helpers/*; do
        [ -f "$source" ] || continue
        relative="''${source#${export}/}"
        target="$destination/$relative"
        # Keep timestamps stable when the generated contents have not changed.
        if [ -f "$target" ] && cmp -s "$source" "$target"; then
          continue
        fi
        temporary=$(mktemp "$target.XXXXXX")
        cp "$source" "$temporary"
        if [ -x "$source" ]; then chmod 755 "$temporary"; else chmod 644 "$temporary"; fi
        mv -f "$temporary" "$target"
      done
      printf 'Work VDI installer ready: %s/install.sh\n' "$destination"
    '';
  };
in
{
  home.packages = [ copy ];

  # Always available, even if this machine does not keep a ~/nix-config checkout.
  xdg.dataFile."work-vdi/export".source = export;

  # The stable data-directory link updates with the generation; refresh the
  # convenient checkout copies too, including direct darwin/home-manager switches.
  home.activation.workVdiExport = lib.hm.dag.entryAfter [ "linkGeneration" ] ''
    if [ -f ${lib.escapeShellArg "${checkout}/flake.nix"} ]; then
      run ${lib.getExe copy} --output ${lib.escapeShellArg "${checkout}/portable/work-vdi"}
    fi
  '';
}
#+end_src
