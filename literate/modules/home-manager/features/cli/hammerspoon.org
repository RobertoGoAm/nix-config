* macOS quake terminal via Hammerspoon — a drop-down Alacritty on...

macOS quake terminal via Hammerspoon — a drop-down Alacritty on Cmd+` (mirroring
the perseus tdrop setup and the keyboard's Cmd/Super+`), plus a transparency
toggle on Cmd+Shift+`. Hammerspoon itself is a Homebrew cask (see the hosts'
casks.nix) and needs a ONE-TIME Accessibility grant (System Settings → Privacy &
Security → Accessibility → Hammerspoon). It replaces the iTerm2 quake, whose
hotkey is disabled in iterm2.nix so Cmd+` is free.

#+begin_src nix
lib.mkIf pkgs.stdenv.hostPlatform.isDarwin {
  home.file.".hammerspoon/init.lua".text = ''
    -- Quake-style drop-down Alacritty on Cmd+`: reveal if hidden, hide if frontmost.
    -- Positioned (dropped from the top) only on FIRST launch; later toggles just
    -- reveal/hide, so it keeps whatever size or fullscreen state you left it in.
    local QUAKE = "Alacritty"

    local function dropTop(win)
      if not win then return end
      local f = win:screen():frame()
      win:setFrame({ x = f.x, y = f.y, w = f.w, h = f.h * 0.45 })
    end

    local function toggleQuake()
      local app = hs.application.get(QUAKE)
      if app and app:isFrontmost() then
        app:hide()
      elseif app and #app:allWindows() > 0 then
        app:activate() -- reveal as-is; don't re-frame, so it keeps its size / fullscreen
      else
        hs.application.launchOrFocus(QUAKE)
        hs.timer.doAfter(0.3, function()
          local a = hs.application.get(QUAKE)
          if a then dropTop(a:mainWindow()) end
        end)
      end
    end

    -- Summon on the TOP-LEFT key (left of 1) regardless of keyboard layout. On ANSI
    -- (Bridge75) that key is ` / grave; on an Apple ISO keyboard the top-left key is
    -- § (keycode 10) while ` sits left-of-Z. Bind both so the same physical key works
    -- on either board (muscle memory). keycode 10 isn't emitted by ANSI boards, so
    -- it's a harmless no-op there.
    -- Cmd+` belongs to OmniWM's quake terminal now, so these are not bound.
    -- Two things answering one key means both fire: the Ghostty panel and an
    -- Alacritty window, every press. `toggleQuake' is left defined and reachable
    -- from the console, which is all it takes to put it back on a key.
    --
    -- Both spellings were bound because the top-left key is ` on the ANSI
    -- Bridge75 and § (keycode 10) on an Apple ISO board; whichever key OmniWM's
    -- own binding ends up on inherits that problem, not this file.
    -- hs.hotkey.bind({ "cmd" }, "`", toggleQuake)
    -- hs.hotkey.bind({ "cmd" }, "#10", toggleQuake)

    -- Toggle terminal transparency on Cmd+Shift+` via Alacritty's runtime IPC.
    local transparent = false
    hs.hotkey.bind({ "cmd", "shift" }, "`", function()
      transparent = not transparent
      hs.execute("alacritty msg config window.opacity=" .. (transparent and "0.85" or "1.0"), true)
    end)

    -- The app keys live in their own file; see apps.lua below.
    require("apps")

    -- Auto-reload this config when ~/.hammerspoon changes (handy while iterating).
    hs.pathwatcher.new(os.getenv("HOME") .. "/.hammerspoon/", hs.reload):start()
    hs.alert.show("Hammerspoon loaded")
  '';
#+end_src

* The app keys the window manager used to own

aerospace carried two bindings that had nothing to do with tiling: =alt-enter=
summoned Emacs and =alt-b= summoned the browser. Turning aerospace off took
both with it, and =alt-enter= is worse than merely gone -- OmniWM binds
=Option+Return= to =toggleFullscreen=, so the old key now maximises whatever is
in front of you.

They come back here because this is the only thing running that can bind a key
to a command at all. OmniWM's hotkeys are a fixed list of 188 window-manager
actions with no case for running something; the nearest it offers is a command
palette.

Emacs keeps =Option+Return=, the key it has always had. OmniWM claimed it for
=toggleFullscreen= and that is what took it away; fullscreen has been moved to
=Option+Shift+Return= instead, which nothing else wanted. The browser sits on
=Option+B=, which OmniWM does not bind either.

Zen is addressed by bundle id rather than by name. The darwin build is the beta
channel and ships as "Zen Browser (Beta).app" while calling itself "Zen", so
neither name is a safe handle and both change with the channel; the id does
not. Chrome is not bound: it exists here for =chrome-dev= and its
=--disable-web-security= profile, which is a thing you open for a local server,
not a thing you summon.

A real toggle, unlike the aerospace version -- that one ran =emacsclient -c -n=
and opened another frame on every press. This hides the app when it is already
in front, raises it when it is running, and only asks for a new frame when
there is none, which is the shape the quake terminal above already has.

emacsclient rather than =open=: Emacs runs as a daemon, so opening the bundle
would start a second, unrelated instance while emacsclient asks the running one
for a frame. =--alternate-editor= with an empty value starts the daemon if
nothing is listening, so the key works before the agent is up. The profile path
rather than the bare name, because by name LaunchServices has picked a stale
Emacs out of an old generation before.

=alt-g= gathers. Zen lives on Media and Obsidian on Productivity, which is
right until the moment you want to read one of them beside what you are
working on. =summon-right= brings a window to the current workspace from
wherever it is -- =assignToWorkspace= is applied when a window opens, not
continuously, so nothing drags it back -- and =rule apply= re-applies the
rules to send it home again, which is why no workspace number appears below.

The layout has to move with them. These workspaces are niri with
=visibleContainerCount = 1=, so gathered windows become more full-width
columns in the scroll and only one is ever on screen; dwindle tiles them
instead. That is per-workspace and reversible, and =alt-shift-l= is the same
toggle by hand.

#+begin_src nix
  home.file.".hammerspoon/apps.lua".text = ''
    local EMACSCLIENT = "/etc/profiles/per-user/${config.home.username}/bin/emacsclient";

    local function toggleApp(name, launch)
      local app = hs.application.get(name)
      if app and app:isFrontmost() then
        app:hide()
      elseif app and #app:allWindows() > 0 then
        app:activate()
      else
        launch()
      end
    end

    hs.hotkey.bind({ "alt" }, "return", function()
      toggleApp("Emacs", function()
        hs.execute(EMACSCLIENT .. " -c -n --alternate-editor= &", true)
      end)
    end)

    hs.hotkey.bind({ "alt" }, "b", function()
      toggleApp("app.zen-browser.zen", function()
        hs.application.launchOrFocusByBundleID("app.zen-browser.zen")
      end)
    end)

    local OMNIWMCTL =
      "${config.programs.omniwm.package}/Applications/OmniWM.app/Contents/MacOS/omniwmctl"

    -- The apps alt-g fetches, by bundle id.
    local GATHER = {
      ["app.zen-browser.zen"] = true,
      ["md.obsidian"] = true,
    }

    local function omni(args)
      return hs.execute(OMNIWMCTL .. " " .. args)
    end

    local function query(what)
      local ok, decoded = pcall(hs.json.decode, omni("query " .. what .. " --format json"))
      if not ok or type(decoded) ~= "table" then return nil end
      local result = decoded.result or {}
      return result.payload
    end

    local function toggleGather()
      local active = query("active-workspace")
      local here = active and active.workspace and active.workspace.displayName
      local windows = query("windows")
      if not here or not windows or not windows.windows then return end

      -- Away if any of them is somewhere else: the press that finds one out of
      -- place is fetching, not sending back.
      local targets, away = {}, false
      for _, w in ipairs(windows.windows) do
        if w.app and GATHER[w.app.bundleId] then
          table.insert(targets, w.id)
          if not (w.workspace and w.workspace.displayName == here) then
            away = true
          end
        end
      end
      if #targets == 0 then return end

      for _, id in ipairs(targets) do
        omni(away and ("window summon-right " .. id) or ("rule apply --window " .. id))
      end
      omni("command set-workspace-layout " .. (away and "dwindle" or "niri"))
    end

    hs.hotkey.bind({ "alt" }, "g", toggleGather)
  '';
}
